package com.example.angonogiants.model;
public class LeaderboardEntry {
    public String playerName;
    public int score;
    public LeaderboardEntry(String playerName, int score) {
        this.playerName = playerName;
        this.score = score;
    }

    public int getScore() {
        return score;
    }

    public String getPlayerName() {
        return playerName;
    }
}
