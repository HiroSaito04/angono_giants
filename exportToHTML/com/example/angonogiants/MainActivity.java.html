package com.example.angonogiants;

import androidx.appcompat.app.AppCompatActivity;
import android.content.Intent;
import android.media.MediaPlayer;
import android.os.Bundle;
import android.util.Log;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;

public class MainActivity extends AppCompatActivity {

    private MediaPlayer bgMusic;
    private MediaPlayer buttonSound;
    private EditText playerNameEditText;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        playerNameEditText = findViewById(R.id.playerNameEditText);

        try {
            bgMusic = MediaPlayer.create(this, R.raw.bgmusic);
            if (bgMusic != null) {
                bgMusic.setLooping(true);
                bgMusic.start();
            } else {
                Log.e("MainActivity", "Failed to create MediaPlayer");
            }
        } catch (Exception e) {
            Log.e("MainActivity", "Error creating MediaPlayer: " + e.getMessage());
        }

        try {
            buttonSound = MediaPlayer.create(this, R.raw.buttonsound);
        } catch (Exception e) {
            Log.e("MainActivity", "Error creating buttonSound: " + e.getMessage());
        }

        Button btnStart = findViewById(R.id.btnStart);
        Button btnLeaderboard = findViewById(R.id.btnLeaderboard);

        btnStart.setOnClickListener(v -> {
            if (bgMusic != null) {
                bgMusic.pause();
            }
            playSound();

            String playerName = playerNameEditText.getText().toString();
            if (playerName.isEmpty()) {
                playerName = "Player";
            }

            Intent intent = new Intent(this, LoadingActivity.class);
            intent.putExtra("playerName", playerName);
            startActivity(intent);
        });

        btnLeaderboard.setOnClickListener(v -> {
            if (bgMusic != null) {
                bgMusic.pause();
            }
            playSound();
            startActivity(new Intent(this, LeaderboardActivity.class));
        });
    }

    private void playSound() {
        if (buttonSound != null) {
            buttonSound.start();
            buttonSound.setOnCompletionListener(new MediaPlayer.OnCompletionListener() {
                @Override
                public void onCompletion(MediaPlayer mp) {
                    mp.seekTo(0);
                }
            });
        }
    }

    @Override
    protected void onResume() {
        super.onResume();
        if (bgMusic != null && !bgMusic.isPlaying()) {
            bgMusic.start();
        }
    }

    @Override
    protected void onPause() {
        super.onPause();
        if (bgMusic != null) {
            bgMusic.pause();
        }
        if (buttonSound != null) {
            buttonSound.pause();
        }
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        if (bgMusic != null) {
            bgMusic.release();
            bgMusic = null;
        }
        if (buttonSound != null) {
            buttonSound.release();
            buttonSound = null;
        }
    }
}
