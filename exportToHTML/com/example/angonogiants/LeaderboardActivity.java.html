package com.example.angonogiants;

import android.content.Context;
import android.content.Intent;
import android.media.MediaPlayer;
import android.os.Bundle;
import android.util.Log;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.ListView;
import android.widget.TextView;

import androidx.appcompat.app.AppCompatActivity;

import com.example.angonogiants.model.LeaderboardEntry;
import com.example.angonogiants.utils.LeaderboardUtils;

import java.util.ArrayList;
import java.util.List;

public class LeaderboardActivity extends AppCompatActivity {

    private MediaPlayer adebayor; // Background music
    private ListView leaderboardListView;
    private List<LeaderboardUtils.ScoreEntry> leaderboardEntries;
    private Button backButton;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_leaderboard);

        leaderboardListView = findViewById(R.id.leaderboardListView);
        backButton = findViewById(R.id.backButton);

        // Load and display scores
        loadScores();
        displayLeaderboard();

        // Set up back button
        backButton.setOnClickListener(v -> {
            stopBackgroundMusic();
            startActivity(new Intent(LeaderboardActivity.this, MainActivity.class));
            finish();
        });

        // Play background music
        playBackgroundMusic();
    }

    private void playBackgroundMusic() {
        try {
            // Ensure MediaPlayer is not already playing
            if (adebayor == null) {
                adebayor = MediaPlayer.create(this, R.raw.adebayor);
                if (adebayor != null) {
                    adebayor.setLooping(true);
                    adebayor.start();
                } else {
                    Log.e("LeaderboardActivity", "MediaPlayer creation failed");
                }
            }
        } catch (Exception e) {
            Log.e("LeaderboardActivity", "Error starting MediaPlayer: " + e.getMessage());
        }
    }

    private void stopBackgroundMusic() {
        if (adebayor != null) {
            adebayor.stop();
            adebayor.release();
            adebayor = null;
        }
    }

    private void loadScores() {
        leaderboardEntries = LeaderboardUtils.getLeaderboard(this);
        if (leaderboardEntries == null) {
            leaderboardEntries = new ArrayList<>(); // Prevent null pointer exception
        }
    }

    private void displayLeaderboard() {
        LeaderboardAdapter adapter = new LeaderboardAdapter(this, leaderboardEntries);
        leaderboardListView.setAdapter(adapter);
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        stopBackgroundMusic();
    }

    private class LeaderboardAdapter extends ArrayAdapter<LeaderboardUtils.ScoreEntry> {
        private Context context;
        private List<LeaderboardUtils.ScoreEntry> leaderboardEntries;

        public LeaderboardAdapter(Context context, List<LeaderboardUtils.ScoreEntry> leaderboardEntries) {
            super(context, android.R.layout.simple_list_item_2, leaderboardEntries);
            this.context = context;
            this.leaderboardEntries = leaderboardEntries;
        }

        @Override
        public View getView(int position, View convertView, ViewGroup parent) {
            if (convertView == null) {
                convertView = LayoutInflater.from(context).inflate(android.R.layout.simple_list_item_2, parent, false);
            }

            TextView playerNameTextView = convertView.findViewById(android.R.id.text1);
            TextView playerScoreTextView = convertView.findViewById(android.R.id.text2);

            LeaderboardUtils.ScoreEntry entry = leaderboardEntries.get(position);
            playerNameTextView.setText(entry.getPlayerName());
            playerScoreTextView.setText(String.valueOf(entry.getScore()));

            return convertView;
        }
    }
}
