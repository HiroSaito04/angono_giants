package com.example.angonogiants;

import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.Canvas;
import android.graphics.Paint;
import android.graphics.Rect;
import android.util.Log;

public class Higante {

    private Bitmap image;
    int x;
    int y;
    private int speed;
    private Rect hitbox;

    public Higante(Context context, int x, int y, int speed) {
        this.image = BitmapFactory.decodeResource(context.getResources(), R.drawable.enemy);
        if (image == null) {
        }
        this.x = x;
        this.y = y;
        this.speed = speed;
        this.hitbox = new Rect(x, y, x + image.getWidth(), y + image.getHeight());
    }
    public void move() {
        y += speed;
        hitbox.offset(0, speed);
    }
    public void draw(Canvas canvas, Paint paint) {
        canvas.drawBitmap(image, x, y, paint);
    }
    public boolean collidesWith(Stall stall) {
        return Rect.intersects(this.hitbox, stall.getHitbox());
    }
    public boolean isTapped(float tapX, float tapY) {
        return tapX >= x && tapX <= x + image.getWidth() &&
                tapY >= y && tapY <= y + image.getHeight();
    }

    public void update() {
    }
}
