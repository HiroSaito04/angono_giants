package com.example.angonogiants.utils;

import android.content.Context;
import android.content.SharedPreferences;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class LeaderboardUtils {

    private static final String LEADERBOARD_PREFS = "Leaderboard";
    private static final String SCORE_KEY_PREFIX = "score_";
    private static final String PLAYER_KEY_PREFIX = "player_";

    public static void saveScore(Context context, String playerName, int score) {
        SharedPreferences sharedPreferences = context.getSharedPreferences(LEADERBOARD_PREFS, Context.MODE_PRIVATE);
        SharedPreferences.Editor editor = sharedPreferences.edit();

        List<ScoreEntry> scores = getSavedScores(context);
        scores.add(new ScoreEntry(playerName, score));

        // Sort scores in descending order
        Collections.sort(scores, Comparator.comparingInt(ScoreEntry::getScore).reversed());

        // Save only the top 10 scores
        int i = 0;
        for (ScoreEntry entry : scores) {
            if (i < 10) {
                editor.putString(PLAYER_KEY_PREFIX + i, entry.playerName);
                editor.putInt(SCORE_KEY_PREFIX + i, entry.score);
                i++;
            }
        }

        editor.putInt("size", i);
        editor.apply();
    }

    private static List<ScoreEntry> getSavedScores(Context context) {
        List<ScoreEntry> scores = new ArrayList<>();
        SharedPreferences sharedPreferences = context.getSharedPreferences(LEADERBOARD_PREFS, Context.MODE_PRIVATE);
        int size = sharedPreferences.getInt("size", 0);

        for (int i = 0; i < size; i++) {
            String playerName = sharedPreferences.getString(PLAYER_KEY_PREFIX + i, "");
            int playerScore = sharedPreferences.getInt(SCORE_KEY_PREFIX + i, 0);
            scores.add(new ScoreEntry(playerName, playerScore));
        }

        return scores;
    }

    public static List<ScoreEntry> getLeaderboard(Context context) {
        return getSavedScores(context);
    }

    public static class ScoreEntry {
        public String playerName;
        public int score;

        public ScoreEntry(String playerName, int score) {
            this.playerName = playerName;
            this.score = score;
        }

        public int getScore() {
            return score;
        }

        public String getPlayerName() { // ✅ FIXED: Return String instead of int
            return playerName;
        }
    }
}
