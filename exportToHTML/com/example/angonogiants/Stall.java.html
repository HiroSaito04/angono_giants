package com.example.angonogiants;

import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.Canvas;
import android.graphics.Paint;
import android.graphics.Rect;
public class Stall {
    private Bitmap image;
    private int x, y;
    private Rect hitbox;

    public Stall(Context context, Bitmap image) {
        this.image = image;
        x = (int) (context.getResources().getDisplayMetrics().widthPixels / 2 - image.getWidth() / 2);
        y = (int) (context.getResources().getDisplayMetrics().heightPixels - image.getHeight() - 50);

        hitbox = new Rect(x, y, x + image.getWidth(), y + image.getHeight());
    }
    public void draw(Canvas canvas, Paint paint) {
        canvas.drawBitmap(image, x, y, paint);
    }
    public Rect getHitbox() {
        return hitbox;
    }
}
