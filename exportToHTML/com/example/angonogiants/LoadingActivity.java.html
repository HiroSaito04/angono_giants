package com.example.angonogiants;

import android.content.Intent;
import android.media.MediaPlayer;
import android.os.Bundle;
import android.os.Handler;
import android.util.Log;
import android.view.View;
import android.widget.Button;
import android.widget.ProgressBar;

import androidx.appcompat.app.AppCompatActivity;

public class LoadingActivity extends AppCompatActivity {

    private ProgressBar progressBar;
    private MediaPlayer lodingloding;
    private Button skipButton;
    private Handler handler;
    private Runnable loadingRunnable;
    private boolean isLoadingComplete = false;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_loading);

        progressBar = findViewById(R.id.progressBar);
        skipButton = findViewById(R.id.skipButton);

        handler = new Handler();

        skipButton.setOnClickListener(new View.OnClickListener() {
            @Override
            public void onClick(View v) {
                isLoadingComplete = true;
                handler.removeCallbacks(loadingRunnable);
                startGameplay();
            }
        });

        startLoading();

        playBackgroundMusic();
    }

    private void playBackgroundMusic() {
        try {
            lodingloding = MediaPlayer.create(this, R.raw.lodingloding);
            if (lodingloding != null) {
                lodingloding.setLooping(true); // Loop background music
                lodingloding.start();
            } else {
                Log.e("LoadingActivity", "Failed to create MediaPlayer");
            }
        } catch (Exception e) {
            Log.e("LoadingActivity", "Error creating MediaPlayer: " + e.getMessage());
        }
    }

    private void startLoading() {
        loadingRunnable = new Runnable() {
            @Override
            public void run() {
                if (!isLoadingComplete) {
                    handler.postDelayed(this, 500);
                }
            }
        };

        handler.postDelayed(loadingRunnable, 500);

        handler.postDelayed(new Runnable() {
            @Override
            public void run() {
                if (!isLoadingComplete) {
                    startGameplay();
                }
            }
        }, 20000);
    }

    private void startGameplay() {
        if (lodingloding != null && lodingloding.isPlaying()) {
            lodingloding.stop();
            lodingloding.release();
            lodingloding = null;
        }

        Intent intent = new Intent(LoadingActivity.this, GameActivity.class);
        intent.putExtra("playerName", getIntent().getStringExtra("playerName"));
        startActivity(intent);
        finish();
    }

    @Override
    protected void onDestroy() {
        super.onDestroy();
        if (lodingloding != null) {
            lodingloding.release();
            lodingloding = null;
        }
        if (handler != null) {
            handler.removeCallbacks(loadingRunnable);
        }
    }
}
