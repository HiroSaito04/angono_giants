package com.example.angonogiants;

import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.Canvas;
import android.graphics.Color;
import android.graphics.Paint;
import android.util.AttributeSet;
import android.view.MotionEvent;
import android.view.View;
import android.widget.Toast;
import java.util.ArrayList;
import java.util.Iterator;
import java.util.List;
import java.util.Random;

public class GameView extends View {

    private List<Enemy> enemies = new ArrayList<>();
    private Paint enemyPaint;
    private Random random = new Random();
    private int screenWidth, screenHeight;
    private int dotsReachedMiddle = 0;
    private final int maxDotsToEndGame = 3;
    private Bitmap backgroundBitmap;
    private boolean gameOver = false;

    public GameView(Context context, AttributeSet attrs) {
        super(context, attrs);
        init();
    }

    private void init() {
        enemyPaint = new Paint();
        enemyPaint.setColor(Color.BLACK);
        enemyPaint.setAntiAlias(true);

        backgroundBitmap = BitmapFactory.decodeResource(getResources(), R.drawable.gmplay);
    }

    @Override
    protected void onSizeChanged(int w, int h, int oldw, int oldh) {
        super.onSizeChanged(w, h, oldw, oldh);
        screenWidth = w;
        screenHeight = h;
        backgroundBitmap = Bitmap.createScaledBitmap(backgroundBitmap, w, h, true);
        spawnEnemies(); // Spawn dots after getting screen size
    }

    private void spawnEnemies() {
        enemies.clear();
        for (int i = 0; i < 30; i++) {
            float x = random.nextInt(screenWidth);
            float y = (i % 2 == 0) ? 0 : screenHeight; // Top or Bottom spawn
            enemies.add(new Enemy(x, y));
        }
    }

    @Override
    protected void onDraw(Canvas canvas) {
        super.onDraw(canvas);

        // Draw background
        if (backgroundBitmap != null) {
            canvas.drawBitmap(backgroundBitmap, 0, 0, null);
        }

        // Draw dots
        for (Enemy enemy : enemies) {
            canvas.drawCircle(enemy.x, enemy.y, 30, enemyPaint);
        }

        // Draw game over message
        if (gameOver) {
            Paint gameOverPaint = new Paint();
            gameOverPaint.setColor(Color.RED);
            gameOverPaint.setTextSize(100);
            canvas.drawText("Game Over!", screenWidth / 4f, screenHeight / 2f, gameOverPaint);
        }
    }

    public void update() {
        if (gameOver) return;

        Iterator<Enemy> iterator = enemies.iterator();
        while (iterator.hasNext()) {
            Enemy enemy = iterator.next();
            enemy.move();

            // Check if enemy reaches the middle
            if (enemy.y >= screenHeight / 2 - 30 && enemy.y <= screenHeight / 2 + 30) {
                dotsReachedMiddle++;
                iterator.remove(); // Remove dot that reached the middle

                if (dotsReachedMiddle >= maxDotsToEndGame) {
                    gameOver = true;
                    postInvalidate();
                    return;
                }
            }
        }
        postInvalidate(); // Redraw screen
    }

    @Override
    public boolean onTouchEvent(MotionEvent event) {
        if (gameOver) return false;

        if (event.getAction() == MotionEvent.ACTION_DOWN) {
            float x = event.getX();
            float y = event.getY();

            Iterator<Enemy> iterator = enemies.iterator();
            while (iterator.hasNext()) {
                Enemy enemy = iterator.next();
                float distance = (float) Math.sqrt(Math.pow(x - enemy.x, 2) + Math.pow(y - enemy.y, 2));
                if (distance < 30) {
                    iterator.remove(); // Remove tapped dot
                    break;
                }
            }
            return true;
        }
        return false;
    }

    public int getScore() {
        return 0;
    }

    private class Enemy {
        float x, y;
        float speedY;

        public Enemy(float x, float y) {
            this.x = x;
            this.y = y;
            this.speedY = (y == 0) ? 5 : -5; // Move downward if top, upward if bottom
        }

        public void move() {
            y += speedY;
        }
    }
}
